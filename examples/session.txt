# Example session: run with
#   java -jar target/order-matching-engine-1.0.0.jar examples/session.txt

# Build a book
sell 100 @ 101.00
sell 50 @ 101.00
sell 70 @ 101.50
buy 40 @ 100.50
buy 25 @ 100.25
book

# An aggressive buy sweeps the first level in time priority (order #1 then #2)
buy 120 @ 101.00
book

# Market order takes the rest of 101.00 then part of 101.50
buy 60 market

# Fill-or-kill that cannot be satisfied is rejected without trading
buy 500 @ 102.00 fok

# Reduce keeps queue priority, replace does not
reduce 4 10
replace 5 25 @ 100.75
book
